import { Injectable, UnauthorizedException } from '@nestjs/common';
import { UsersService } from '../users/users.service';
import { JwtService } from '@nestjs/jwt';
import * as bcrypt from 'bcrypt';

@Injectable()
export class AuthService {
  constructor(
    private usersService: UsersService,
    private jwtService: JwtService
  ) {}

  async validateUser(email: string, pass: string): Promise<any> {
    const user = await this.usersService.findOne(email);
    if (user && (await bcrypt.compare(pass, user.password))) {
      const { password, ...result } = user;
      return result;
    }
    return null;
  }

  async login(user: any) {
    const payload = { email: user.email, sub: user.id, role: user.role };
    return {
      access_token: this.jwtService.sign(payload),
      user: {
          id: user.id,
          email: user.email,
          role: user.role,
          kycStatus: user.kyc?.status || 'NOT_SUBMITTED'
      }
    };
  }

  async signup(data: any) {
    const user = await this.usersService.createUser(data);
    return this.login(user);
  }
}

import { Injectable, BadRequestException, Logger } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { WalletService } from '../wallet/wallet.service';
import { BinarySide, BinaryStatus, LedgerType } from '@prisma/client';
import { Cron, CronExpression } from '@nestjs/schedule';
import { BinaryGateway } from './binary.gateway';

@Injectable()
export class BinaryService {
  private readonly logger = new Logger(BinaryService.name);

  constructor(
    private prisma: PrismaService,
    private walletService: WalletService,
    private binaryGateway: BinaryGateway,
  ) {}

  // Open a new position
  async openPosition(userId: string, pair: string, amount: number, side: BinarySide, duration: number) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');
    if (![30, 60, 120, 300].includes(duration)) throw new BadRequestException('Invalid duration');

    // 1. Lock Funds (Investment Amount)
    // Binary locks the investment. If you lose, it's gone. If you win, you get Amount + Profit.
    // We lock 'USDT' (Quote currency usually)
    const currency = 'USDT'; 
    await this.walletService.lockFunds(userId, currency, amount);

    // 2. Get Current Price (Mock for now, real implementation needs Live Price Service)
    const entryPrice = await this.getCurrentPrice(pair);

    // 3. Create Position
    const expiresAt = new Date(Date.now() + duration * 1000);
    const payoutRate = 0.85; // 85% profit example
    const payout = amount * payoutRate;

    const position = await this.prisma.binaryOptionPosition.create({
      data: {
        userId,
        pair,
        amount,
        side,
        entryPrice,
        payout,
        duration,
        expiresAt,
        status: BinaryStatus.OPEN,
      },
    });

    // Notify User via WS
    this.binaryGateway.server.to(userId).emit('binary_position_opened', position);

    return position;
  }

  // Check for expired positions (Runs every second)
  @Cron(CronExpression.EVERY_SECOND)
  async checkExpiredPositions() {
    const now = new Date();
    
    // Find open positions that have expired
    const expiredPositions = await this.prisma.binaryOptionPosition.findMany({
      where: {
        status: BinaryStatus.OPEN,
        expiresAt: { lte: now },
      },
    });

    for (const position of expiredPositions) {
      await this.settlePosition(position);
    }
  }

  async settlePosition(position: any) {
    // Get Close Price
    const closePrice = await this.getCurrentPrice(position.pair);
    
    // Determine Outcome
    let status: BinaryStatus = BinaryStatus.LOST;
    let profit = -position.amount; // Default Loss
    
    if (position.side === BinarySide.CALL) {
      if (closePrice > position.entryPrice) status = BinaryStatus.WON;
      else if (closePrice === position.entryPrice) status = BinaryStatus.DRAW;
    } else { // PUT
      if (closePrice < position.entryPrice) status = BinaryStatus.WON;
      else if (closePrice === position.entryPrice) status = BinaryStatus.DRAW;
    }

    // Payout Logic
    if (status === BinaryStatus.WON) {
      profit = position.payout;
      // Return Investment + Profit
      const totalReturn = position.amount + position.payout;
      
      // 1. Deduct Locked (The investment is currently locked)
      await this.walletService.deductLockedFunds(position.userId, 'USDT', position.amount, LedgerType.TRADE_BINARY, `BINARY-UNLOCK-${position.id}`);
      // 2. Add Total Return (Investment + Profit)
      // Actually, cleaner: Deduct locked (burn it), then Add Total Return as Deposit?
      // Or: Unlock investment, Add Profit?
      // Let's stick to: Deduct Locked (Investment used), Add Winnings.
      // Wait, if I deduct locked, the user loses the investment from their balance.
      // So I need to add back: Investment + Profit.
      await this.walletService.deposit(position.userId, 'USDT', totalReturn, `BINARY-WIN-${position.id}`);
      
    } else if (status === BinaryStatus.DRAW) {
      profit = 0;
      // Return Investment
      await this.walletService.unlockFunds(position.userId, 'USDT', position.amount);
      
    } else {
      // LOST
      // Deduct Locked Funds (User loses investment)
      await this.walletService.deductLockedFunds(position.userId, 'USDT', position.amount, LedgerType.TRADE_BINARY, `BINARY-LOSS-${position.id}`);
    }

    // Update DB
    await this.prisma.$transaction([
      this.prisma.binaryOptionPosition.update({
        where: { id: position.id },
        data: { status, strikePrice: closePrice },
      }),
      this.prisma.binaryOptionResult.create({
        data: {
          positionId: position.id,
          closePrice,
          profit,
        },
      }),
    ]);

    // Notify User via WS
    this.binaryGateway.server.to(position.userId).emit('binary_position_closed', {
      id: position.id,
      status,
      closePrice,
      profit,
    });
    
    this.logger.log(`Position ${position.id} settled: ${status}`);
  }

  // Admin Override Outcome
  async adminForceResult(positionId: string, outcome: 'WIN' | 'LOSE') {
      const position = await this.prisma.binaryOptionPosition.findUnique({ where: { id: positionId } });
      if (!position || position.status !== BinaryStatus.OPEN) throw new BadRequestException('Invalid position');

      // Mock price to force outcome
      // If force WIN and CALL -> price = entry + 1
      // If force LOSE and CALL -> price = entry - 1
      let forcedPrice = position.entryPrice;
      if (outcome === 'WIN') {
          forcedPrice = position.side === BinarySide.CALL ? position.entryPrice + 100 : position.entryPrice - 100;
      } else {
          forcedPrice = position.side === BinarySide.CALL ? position.entryPrice - 100 : position.entryPrice + 100;
      }

      // We need to pass this price to settlePosition. 
      // But settlePosition fetches price internally.
      // Refactor: Allow passing price to settlePosition or creating a separate method.
      // For now, let's just duplicate logic slightly or update the implementation.
      
      // Hack for MVP: Update `getCurrentPrice` to check for overrides? No, that's messy.
      // Better: Create `settlePositionWithPrice`.
      
      await this.settlePositionWithPrice(position, forcedPrice);
      return { message: `Position forced to ${outcome}` };
  }

  async settlePositionWithPrice(position: any, closePrice: number) {
      // ... Copy logic from settlePosition but use provided closePrice ...
      // For DRY, I'll extract the core logic later. 
      // For this step, I'll just implement the logic directly to ensure it works.
      
      let status: BinaryStatus = BinaryStatus.LOST;
      let profit = -position.amount;
      
      if (position.side === BinarySide.CALL) {

        else if (closePrice === position.entryPrice) status = BinaryStatus.DRAW;
      } else {
        if (closePrice < position.entryPrice) status = BinaryStatus.WON;
        else if (closePrice === position.entryPrice) status = BinaryStatus.DRAW;
      }
      
      if (status === BinaryStatus.WON) {
          profit = position.payout;
          const totalReturn = position.amount + position.payout;
          await this.walletService.deductLockedFunds(position.userId, 'USDT', position.amount, LedgerType.TRADE_BINARY, `BINARY-UNLOCK-${position.id}`);
          await this.walletService.deposit(position.userId, 'USDT', totalReturn, `BINARY-WIN-${position.id}`);
      } else if (status === BinaryStatus.DRAW) {
          profit = 0;
          await this.walletService.unlockFunds(position.userId, 'USDT', position.amount);
      } else {
          await this.walletService.deductLockedFunds(position.userId, 'USDT', position.amount, LedgerType.TRADE_BINARY, `BINARY-LOSS-${position.id}`);

import { PrismaService } from '../prisma/prisma.service';
import { WalletService } from '../wallet/wallet.service';
import { LedgerType } from '@prisma/client';

@Injectable()
export class ConversionService {
  // Mock rates for MVP. In prod, fetch from live API or DB settings.
  private rates: Record<string, number> = {
    'USDT:BTC': 0.000025, // 1 USDT = 0.000025 BTC
    'BTC:USDT': 40000,    // 1 BTC = 40000 USDT
    'USDT:ETH': 0.0005,
    'ETH:USDT': 2000,
  };

  constructor(
    private prisma: PrismaService,
    private walletService: WalletService,
  ) {}

  async getRate(from: string, to: string): Promise<number> {
    const key = `${from}:${to}`;
    return this.rates[key] || 0;
  }

  async convert(userId: string, fromCoin: string, toCoin: string, amount: number) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');
    
    const rate = await this.getRate(fromCoin, toCoin);
    if (!rate || rate === 0) throw new BadRequestException(`Conversion from ${fromCoin} to ${toCoin} not supported`);

    const receiveAmount = amount * rate;

    // Transactional Conversion
    // 1. Withdraw 'fromCoin'
    await this.walletService.withdraw(userId, fromCoin, amount, `CONVERT-OUT-${Date.now()}`);
    
    // 2. Deposit 'toCoin'
    await this.walletService.deposit(userId, toCoin, receiveAmount, `CONVERT-IN-${Date.now()}`);

    // 3. Log Transaction
    return this.prisma.conversionTransaction.create({
      data: {
        userId,
        fromCoin,
        toCoin,
        amount,
        rate,
        received: receiveAmount,
      },
    });
  }

  // Admin: Update Rate
  async updateRate(pair: string, newRate: number) {
      this.rates[pair] = newRate;
      return { message: `Rate for ${pair} updated to ${newRate}` };
  }
}

import { Injectable, BadRequestException, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { WalletService } from '../wallet/wallet.service';
import { LoanStatus, LedgerType } from '@prisma/client';

@Injectable()
export class LoanService {
  constructor(
    private prisma: PrismaService,
    private walletService: WalletService,
  ) {}

  // Apply for a loan
  async applyForLoan(userId: string, amount: number, durationDays: number, collateralImg: string, collateralDesc: string) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');

    // Simple logic: Interest is 1% per day for MVP
    const interestRate = 0.01;
    const totalRepayment = amount + (amount * interestRate * durationDays);

    return this.prisma.loanApplication.create({
      data: {
        userId,
        amount,
        currency: 'USDT', // Default to USDT loans
        durationDays,
        interestRate,
        totalRepayment,
        collateral: collateralDesc,
        collateralImg,
        status: LoanStatus.PENDING,
      },
    });
  }

  // Admin: Approve Loan
  async approveLoan(loanId: string) {
    const loan = await this.prisma.loanApplication.findUnique({ where: { id: loanId } });
    if (!loan) throw new NotFoundException('Loan not found');
    if (loan.status !== LoanStatus.PENDING) throw new BadRequestException('Loan is not pending');

    // 1. Credit User Wallet with Loan Amount
    await this.walletService.deposit(loan.userId, loan.currency, loan.amount, `LOAN-DISBURSE-${loan.id}`);

    // 2. Update Loan Status
    return this.prisma.loanApplication.update({
      where: { id: loanId },
      data: {
        status: LoanStatus.ACTIVE,
        approvedAt: new Date(),
        dueDate: new Date(Date.now() + loan.durationDays * 24 * 60 * 60 * 1000),
      },
    });
  }

  // Admin: Reject Loan
  async rejectLoan(loanId: string) {
    const loan = await this.prisma.loanApplication.findUnique({ where: { id: loanId } });
    if (!loan) throw new NotFoundException('Loan not found');
    if (loan.status !== LoanStatus.PENDING) throw new BadRequestException('Loan is not pending');

    return this.prisma.loanApplication.update({
      where: { id: loanId },
      data: { status: LoanStatus.REJECTED },
    });
  }

  // Get User Loans
  async getUserLoans(userId: string) {
    return this.prisma.loanApplication.findMany({
      where: { userId },
      orderBy: { createdAt: 'desc' },
    });
  }

  // Get All Loans (Admin)
  async getAllLoans() {
    return this.prisma.loanApplication.findMany({
      orderBy: { createdAt: 'desc' },
      include: { user: { select: { email: true } } }
    });
  }

import { MiningStatus, LedgerType } from '@prisma/client';
import { Cron, CronExpression } from '@nestjs/schedule';

@Injectable()
export class MiningService {
  private readonly logger = new Logger(MiningService.name);

  constructor(
    private prisma: PrismaService,
    private walletService: WalletService,
  ) {}

  // Admin: Create Plan
  async createPlan(name: string, dailyYield: number, lockupPeriod: number, minDeposit: number, maxDeposit: number) {
    return this.prisma.miningPlan.create({
      data: {
        name,
        dailyYield,
        lockupPeriod,
        minDeposit,
        maxDeposit,
        currency: 'USDT',
      },
    });
  }
  
  async getPlans() {
      return this.prisma.miningPlan.findMany({ where: { isActive: true } });
  }

  // User: Subscribe
  async subscribe(userId: string, planId: string, amount: number) {
    const plan = await this.prisma.miningPlan.findUnique({ where: { id: planId } });
    if (!plan) throw new NotFoundException('Plan not found');
    if (!plan.isActive) throw new BadRequestException('Plan is inactive');
    if (amount < plan.minDeposit || amount > plan.maxDeposit) {
      throw new BadRequestException(`Amount must be between ${plan.minDeposit} and ${plan.maxDeposit}`);
    }

    // Lock Funds
    await this.walletService.lockFunds(userId, 'USDT', amount);

    const endDate = new Date();
    endDate.setDate(endDate.getDate() + plan.lockupPeriod);

    return this.prisma.miningSubscription.create({
      data: {
        userId,
        planId,
        amount,
        startDate: new Date(),
        endDate,
        status: MiningStatus.ACTIVE,
      },
    });
  }

  // Cron: Distribute Daily Yield
  @Cron(CronExpression.EVERY_DAY_AT_MIDNIGHT)
  async distributeRewards() {
    this.logger.log('Starting daily yield distribution...');
    const activeSubs = await this.prisma.miningSubscription.findMany({
      where: { status: MiningStatus.ACTIVE },
      include: { plan: true },
    });

    for (const sub of activeSubs) {
      // Check if expired
      if (new Date() > sub.endDate) {
        await this.completeSubscription(sub);
        continue;
      }

      // Calculate Reward
      const reward = sub.amount * (sub.plan.dailyYield / 100);
      
      // Credit Reward to Wallet
      await this.walletService.deposit(sub.userId, 'USDT', reward, `MINING-YIELD-${sub.id}`);

      // Update Subscription Stats
      await this.prisma.miningSubscription.update({
        where: { id: sub.id },
        data: {
          totalRewards: { increment: reward },
          lastRewardPaid: new Date(),
        },
      });
    }
    this.logger.log(`Distributed rewards to ${activeSubs.length} subscriptions.`);
  }

  async completeSubscription(sub: any) {
    // Return Principal
    // The principal was locked. We need to unlock it.
    // Or better: Deduct Locked, Credit Available (like a transfer back).
    
    // Step 1: Deduct from Locked (Burn the reservation)
    await this.walletService.deductLockedFunds(sub.userId, 'USDT', sub.amount, LedgerType.MINING_REWARD, `MINING-UNLOCK-${sub.id}`);
    
    // Step 2: Credit Principal back to Available
    await this.walletService.deposit(sub.userId, 'USDT', sub.amount, `MINING-RETURN-${sub.id}`);

    await this.prisma.miningSubscription.update({
      where: { id: sub.id },
      data: { status: MiningStatus.COMPLETED },
    });
    
    this.logger.log(`Subscription ${sub.id} completed.`);
  }

  async getUserSubscriptions(userId: string) {
      return this.prisma.miningSubscription.findMany({
          where: { userId },
          include: { plan: true }
      });
  }
}

import { Injectable, BadRequestException, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { WalletService } from '../wallet/wallet.service';
import { OrderSide, OrderType, OrderStatus, LedgerType } from '@prisma/client';

@Injectable()
export class SpotService {
  constructor(
    private prisma: PrismaService,
    private walletService: WalletService,
  ) {}

  async placeOrder(userId: string, pair: string, side: OrderSide, type: OrderType, amount: number, price?: number) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');
    if (type === OrderType.LIMIT && (!price || price <= 0)) throw new BadRequestException('Price is required for Limit orders');

    // 1. Calculate required funds to lock
    let requiredFunds = 0;
    let currencyToLock = '';

    const [base, quote] = pair.split('/'); // e.g., BTC/USDT -> Base: BTC, Quote: USDT

    if (side === OrderSide.BUY) {
      // Buying BTC with USDT
      currencyToLock = quote;
      if (type === OrderType.LIMIT) {
        requiredFunds = amount * price!;
      } else {
        // Market Buy: For simplicity in this MVP, we might require an estimated amount or lock a large chunk.
        // Better approach for Market Buy is "Spend X USDT" rather than "Buy Y BTC" to avoid calculation issues.
        // Assuming 'amount' for Market Buy is 'quote currency amount' (Spend 100 USDT).
        // However, standard is usually "Buy 1 BTC".
        // Let's assume Limit orders for MVP core or estimate market price.
        // For robustness, let's stick to LIMIT orders primarily or strictly defined Market logic.
        // Let's implement LIMIT first for safety.
        if (!price) throw new BadRequestException('Market orders require current price estimation (not fully implemented in MVP step 1)');
        requiredFunds = amount * price; 
      }
    } else {
      // Selling BTC for USDT
      currencyToLock = base;
      requiredFunds = amount;
    }

    // 2. Lock Funds
    await this.walletService.lockFunds(userId, currencyToLock, requiredFunds);

    // 3. Create Order
    const order = await this.prisma.spotOrder.create({
      data: {
        userId,
        pair,
        side,
        type,
        price,
        amount,
        remaining: amount,
        status: OrderStatus.OPEN,
      },
    });

    // 4. Trigger Matching (Async)
    this.matchOrders(pair);

    return order;
  }

  async matchOrders(pair: string) {
    // Simple matching engine for MVP
    // Match Buy Limit vs Sell Limit
    
    // Fetch open buy orders sorted by Price DESC (Highest bid first)
    const buyOrders = await this.prisma.spotOrder.findMany({
      where: { pair, side: OrderSide.BUY, status: { in: [OrderStatus.OPEN, OrderStatus.PARTIALLY_FILLED] } },
      orderBy: { price: 'desc' },
    });

    // Fetch open sell orders sorted by Price ASC (Lowest ask first)
    const sellOrders = await this.prisma.spotOrder.findMany({
      where: { pair, side: OrderSide.SELL, status: { in: [OrderStatus.OPEN, OrderStatus.PARTIALLY_FILLED] } },
      orderBy: { price: 'asc' },
    });

    for (const buyOrder of buyOrders) {
      for (const sellOrder of sellOrders) {
        if (!buyOrder.price || !sellOrder.price) continue; // Skip market orders for now
        
        // Check if match is possible
        if (buyOrder.price >= sellOrder.price) {
          // Match found!
          const matchPrice = sellOrder.price; // Usually executes at maker price
          const matchAmount = Math.min(buyOrder.remaining, sellOrder.remaining);

          if (matchAmount <= 0) continue;

          // Execute Trade
          await this.executeTrade(buyOrder, sellOrder, matchPrice, matchAmount);

          // Update local remaining for loop continuity
          buyOrder.remaining -= matchAmount;
          sellOrder.remaining -= matchAmount;

          if (buyOrder.remaining === 0) break; // Buy order filled, move to next buy order
        }
      }
    }
  }

  async executeTrade(buyOrder: any, sellOrder: any, price: number, amount: number) {
    await this.prisma.$transaction(async (tx) => {
      // 1. Create Trade Record
      const trade = await tx.spotTrade.create({
        data: {
          orderId: buyOrder.id, // Linking to buy order for now, theoretically should link both or have 2 entries
          userId: buyOrder.userId, // Buyer
          pair: buyOrder.pair,
          price,
          amount,
          side: OrderSide.BUY,
          fee: 0, // 0 fee for now
          feeCurrency: 'USDT',
        }
      });
      // Need a second trade record for the seller? 
      // Typically an exchange creates one "Trade" entity linking maker and taker, or two records.
      // Our schema has `orderId` and `userId`. Let's create two records for clarity.
      await tx.spotTrade.create({
        data: {
          orderId: sellOrder.id,
          userId: sellOrder.userId, // Seller
          pair: sellOrder.pair,
          price,
          amount,
          side: OrderSide.SELL,
          fee: 0,
          feeCurrency: 'USDT',
        }
      });

      // 2. Update Orders
      await this.updateOrderStatus(tx, buyOrder.id, amount);
      await this.updateOrderStatus(tx, sellOrder.id, amount);

      // 3. Settlement
      const [base, quote] = buyOrder.pair.split('/');
      const totalQuote = amount * price;

      // Buyer:
      // Locked USDT was `amount * buyOrder.price` (max willingness).
      // Actual cost is `amount * price`.
      // If buyOrder.price > price, refund the difference.
      const lockedQuote = amount * buyOrder.price;
      const actualCost = totalQuote;
      
      // Deduct used USDT from locked
      // We need to access WalletService from within transaction... 
      // Ideally WalletService methods should accept a prisma transaction client.
      // For MVP, we will use the injected service which uses a new transaction.
      // This breaks atomicity of the whole trade execution slightly but is acceptable for this level.
      // TODO: Refactor WalletService to accept `tx`.

      // BUYER Settlement:
      // Deduct Cost from Locked USDT
      // If partial fill, we only deduct for the amount filled.
      // NOTE: WalletService `deductLockedFunds` assumes we deduct what we used.
      // But wait, we locked `buyOrder.price * amount`. We spent `price * amount`.
      // We should deduct `price * amount` from LOCKED? No, that leaves dust in locked.
      // We should deduct `buyOrder.price * amount` from LOCKED, and credit back `(buyOrder.price - price) * amount` to AVAILABLE?
      // Or simply: Unlock everything for this portion, then Deduct actual cost.
      
      // Simpler approach for MVP:
      // Buyer gets BASE currency (creditUser)
      // Seller gets QUOTE currency (creditUser)
      // Buyer loses QUOTE currency (deductLocked)
      // Seller loses BASE currency (deductLocked)
      
      // BUYER: Receives Base
      // We can't call this.walletService.creditUser inside tx easily without refactor.
      // Let's do it after or use `prisma` directly here for speed.
      
      // Update Wallets manually in this TX for atomicity
      
      // BUYER:
      // 1. Unlock the portion of USDT reserved for this fill
      // 2. Deduct the actual USDT cost
      // 3. Add the BTC bought
      
      // SELLER:
      // 1. Deduct the BTC locked
      // 2. Add the USDT earned
    });
    
    // Execute Wallet Changes (Sequential for safety if not in same TX, or refactor)
    // For this MVP, I will call WalletService methods sequentially.
    // Ideally these should be in the transaction above.
    
    const [base, quote] = buyOrder.pair.split('/');
    const totalQuote = amount * price;

    // BUYER Settlement
    // 1. Deduct USDT (Cost)
    // We locked `amount * buyOrder.price`.
    // We need to deduct `amount * buyOrder.price` from locked to clear the reservation
    // But we only actually spent `amount * price`.
    // So we credit back `amount * (buyOrder.price - price)` to available?
    // Let's simplify:
    // Deduct full reserved amount for this chunk from Locked.
    // Credit difference to Available.
    // Credit Bought Asset to Available.
    
    // Deduction: 
    // We locked funds based on Order Price.
    await this.walletService.deductLockedFunds(buyOrder.userId, quote, amount * buyOrder.price, LedgerType.TRADE_SPOT, `TRADE-${buyOrder.id}`);
    // Refund difference (if bought cheaper)
    if (buyOrder.price > price) {
        const refund = amount * (buyOrder.price - price);
        if (refund > 0) {
             await this.walletService.deposit(buyOrder.userId, quote, refund, `REFUND-${buyOrder.id}`);
        }
    }
    // Credit Asset
    await this.walletService.deposit(buyOrder.userId, base, amount, `TRADE-RECV-${buyOrder.id}`);


    // SELLER Settlement
    // 1. Deduct Locked Base (BTC)
    await this.walletService.deductLockedFunds(sellOrder.userId, base, amount, LedgerType.TRADE_SPOT, `TRADE-${sellOrder.id}`);
    // 2. Credit Quote (USDT)
    await this.walletService.deposit(sellOrder.userId, quote, totalQuote, `TRADE-RECV-${sellOrder.id}`);
  }

  async updateOrderStatus(tx: any, orderId: string, amountFilled: number) {
    const order = await tx.spotOrder.findUnique({ where: { id: orderId } });
    const newFilled = order.filled + amountFilled;
    const newRemaining = order.remaining - amountFilled;
    const status = newRemaining <= 0.00000001 ? OrderStatus.FILLED : OrderStatus.PARTIALLY_FILLED;

    await tx.spotOrder.update({
      where: { id: orderId },
      data: {
        filled: newFilled,
        remaining: newRemaining,
        status,
      },
    });
  }

  async cancelOrder(userId: string, orderId: string) {
    const order = await this.prisma.spotOrder.findUnique({ where: { id: orderId } });
    if (!order) throw new NotFoundException('Order not found');
    if (order.userId !== userId) throw new BadRequestException('Not your order');
    if (order.status === OrderStatus.FILLED || order.status === OrderStatus.CANCELLED) {
      throw new BadRequestException('Order cannot be cancelled');
    }

    // Update Status
    await this.prisma.spotOrder.update({
      where: { id: orderId },
      data: { status: OrderStatus.CANCELLED },
    });

    // Unlock Remaining Funds
    const [base, quote] = order.pair.split('/');
    if (order.side === OrderSide.BUY) {
      // Unlock USDT
      // Locked amount was remaining * price
      const lockedAmount = order.remaining * (order.price || 0);
      await this.walletService.unlockFunds(userId, quote, lockedAmount);
    } else {
      // Unlock BTC
      await this.walletService.unlockFunds(userId, base, order.remaining);
    }

    return { message: 'Order cancelled' };
  }

  async getOrderBook(pair: string) {
      const buys = await this.prisma.spotOrder.findMany({
          where: { pair, side: OrderSide.BUY, status: { in: [OrderStatus.OPEN, OrderStatus.PARTIALLY_FILLED] } },
          orderBy: { price: 'desc' },
          take: 50

import { Injectable, NotFoundException, BadRequestException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { LedgerType } from '@prisma/client';

@Injectable()
export class WalletService {
  constructor(private prisma: PrismaService) {}

  // Initialize wallet for a new user
  async createWallet(userId: string) {
    return this.prisma.wallet.create({
      data: {
        userId,
        balances: {}, // Initialize empty JSON object
      },
    });
  }

  // Get user balance for a specific currency
  async getBalance(userId: string, currency: string) {
    const wallet = await this.prisma.wallet.findUnique({
      where: { userId },
    });

    if (!wallet) throw new NotFoundException('Wallet not found');

    const balances = wallet.balances as Record<string, { available: number; locked: number }>;
    return balances[currency] || { available: 0, locked: 0 };
  }

  // Get all balances
  async getAllBalances(userId: string) {
    const wallet = await this.prisma.wallet.findUnique({
      where: { userId },
    });
    if (!wallet) throw new NotFoundException('Wallet not found');
    return wallet.balances;
  }

  // Deposit funds (Admin or Simulated)
  async deposit(userId: string, currency: string, amount: number, referenceId?: string) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');

    return this.prisma.$transaction(async (tx) => {
      const wallet = await tx.wallet.findUnique({ where: { userId } });
      if (!wallet) throw new NotFoundException('Wallet not found');

      const balances = wallet.balances as Record<string, { available: number; locked: number }>;
      const current = balances[currency] || { available: 0, locked: 0 };

      // Update balance
      current.available += amount;
      balances[currency] = current;

      // Update wallet
      await tx.wallet.update({
        where: { userId },
        data: { balances },
      });

      // Create Ledger Entry
      await tx.ledger.create({
        data: {
          walletId: wallet.id,
          currency,
          amount,
          type: LedgerType.DEPOSIT,
          referenceId: referenceId || `DEP-${Date.now()}`,
        },
      });

      return balances[currency];
    });
  }

  // Withdraw funds
  async withdraw(userId: string, currency: string, amount: number, referenceId?: string) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');

    return this.prisma.$transaction(async (tx) => {
      const wallet = await tx.wallet.findUnique({ where: { userId } });
      if (!wallet) throw new NotFoundException('Wallet not found');

      const balances = wallet.balances as Record<string, { available: number; locked: number }>;
      const current = balances[currency] || { available: 0, locked: 0 };

      if (current.available < amount) {
        throw new BadRequestException('Insufficient funds');
      }

      // Update balance
      current.available -= amount;
      balances[currency] = current;

      await tx.wallet.update({
        where: { userId },
        data: { balances },
      });

      // Create Ledger Entry
      await tx.ledger.create({
        data: {
          walletId: wallet.id,
          currency,
          amount: -amount, // Negative for withdrawal
          type: LedgerType.WITHDRAWAL,
          referenceId: referenceId || `WTH-${Date.now()}`,
        },
      });

      return balances[currency];
    });
  }

  // Lock funds (for trading)
  async lockFunds(userId: string, currency: string, amount: number) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');

    return this.prisma.$transaction(async (tx) => {
      const wallet = await tx.wallet.findUnique({ where: { userId } });
      if (!wallet) throw new NotFoundException('Wallet not found');

      const balances = wallet.balances as Record<string, { available: number; locked: number }>;
      const current = balances[currency] || { available: 0, locked: 0 };

      if (current.available < amount) {
        throw new BadRequestException('Insufficient available funds to lock');
      }

      current.available -= amount;
      current.locked += amount;
      balances[currency] = current;

      await tx.wallet.update({
        where: { userId },
        data: { balances },
      });

      return true;
    });
  }

  // Unlock funds (e.g. order cancelled)
  async unlockFunds(userId: string, currency: string, amount: number) {
    if (amount <= 0) throw new BadRequestException('Amount must be positive');

    return this.prisma.$transaction(async (tx) => {
      const wallet = await tx.wallet.findUnique({ where: { userId } });
      if (!wallet) throw new NotFoundException('Wallet not found');

      const balances = wallet.balances as Record<string, { available: number; locked: number }>;
      const current = balances[currency] || { available: 0, locked: 0 };

      if (current.locked < amount) {
        throw new BadRequestException('Insufficient locked funds to unlock');
      }

      current.locked -= amount;
      current.available += amount;
      balances[currency] = current;

      await tx.wallet.update({
        where: { userId },
        data: { balances },
      });

      return true;
    });
  }

  // Deduct locked funds (e.g. order filled or trade lost)
  async deductLockedFunds(userId: string, currency: string, amount: number, type: LedgerType, refId: string) {
     return this.prisma.$transaction(async (tx) => {
      const wallet = await tx.wallet.findUnique({ where: { userId } });
      if (!wallet) throw new NotFoundException('Wallet not found');

      const balances = wallet.balances as Record<string, { available: number; locked: number }>;
      const current = balances[currency] || { available: 0, locked: 0 };

      if (current.locked < amount) {
         // Fallback if something went wrong, but ideally shouldn't happen
         if (current.available + current.locked >= amount) {
             // Deduct from available if locked is missing (edge case)
             const remaining = amount - current.locked;
             current.locked = 0;
             current.available -= remaining;
         } else {
             throw new BadRequestException('Insufficient funds');
         }
      } else {
          current.locked -= amount;
      }
      
      balances[currency] = current;

      await tx.wallet.update({
        where: { userId },
        data: { balances },
      });

      await tx.ledger.create({
        data: {
          walletId: wallet.id,
          currency,
          amount: -amount,
          type: type,
          referenceId: refId,
        },
      });

      return true;
    });
  }

'use client';

import { useState, useEffect } from 'react';

interface CryptoPrice {
  symbol: string;
  name: string;
  price: number;
  change: number;
  icon: string;
}

export function useCryptoPrices() {
  const [prices, setPrices] = useState<CryptoPrice[]>([
    { symbol: 'BTC', name: 'Bitcoin', price: 0, change: 0, icon: '₿' },
    { symbol: 'ETH', name: 'Ethereum', price: 0, change: 0, icon: 'Ξ' },
    { symbol: 'LTC', name: 'Litecoin', price: 0, change: 0, icon: 'Ł' },
    { symbol: 'XRP', name: 'Ripple', price: 0, change: 0, icon: '✕' },
    { symbol: 'BCH', name: 'Bitcoin Cash', price: 0, change: 0, icon: '฿' },
  ]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchPrices = async () => {
      try {
        const symbols = [
          { symbol: 'BTCUSDT', name: 'Bitcoin', displaySymbol: 'BTC', icon: '₿' },
          { symbol: 'ETHUSDT', name: 'Ethereum', displaySymbol: 'ETH', icon: 'Ξ' },
          { symbol: 'LTCUSDT', name: 'Litecoin', displaySymbol: 'LTC', icon: 'Ł' },
          { symbol: 'XRPUSDT', name: 'Ripple', displaySymbol: 'XRP', icon: '✕' },
          { symbol: 'BCHUSDT', name: 'Bitcoin Cash', displaySymbol: 'BCH', icon: '฿' },
        ];

        const pricePromises = symbols.map(async (crypto) => {
          try {
            const tickerResponse = await fetch(
              `https://api.binance.com/api/v3/ticker/24hr?symbol=${crypto.symbol}`
            );
            
            if (!tickerResponse.ok) {
              throw new Error(`Failed to fetch ${crypto.symbol}`);
            }

            const tickerData = await tickerResponse.json();
            
            return {
              symbol: crypto.displaySymbol,
              name: crypto.name,
              price: parseFloat(tickerData.lastPrice),
              change: parseFloat(tickerData.priceChangePercent),
              icon: crypto.icon,
            };
          } catch (error) {
            console.error(`Error fetching ${crypto.symbol}:`, error);
            return null;
          }
        });

        const results = await Promise.all(pricePromises);
        const validPrices = results.filter((p): p is CryptoPrice => p !== null);

        if (validPrices.length > 0) {
          console.log('Binance API prices:', validPrices);
          setPrices(validPrices);
          setLoading(false);
        } else {
          throw new Error('No valid prices fetched');
        }
      } catch (error) {
        console.error('Error fetching crypto prices:', error);
        const fallbackPrices: CryptoPrice[] = [
          { symbol: 'BTC', name: 'Bitcoin', price: 102500, change: 2.5, icon: '₿' },
          { symbol: 'ETH', name: 'Ethereum', price: 3850, change: 1.8, icon: 'Ξ' },
          { symbol: 'LTC', name: 'Litecoin', price: 125, change: -0.5, icon: 'Ł' },
          { symbol: 'XRP', name: 'Ripple', price: 2.85, change: 3.2, icon: '✕' },
          { symbol: 'BCH', name: 'Bitcoin Cash', price: 485, change: -1.2, icon: '฿' },
        ];
        setPrices(fallbackPrices);
        setLoading(false);
      }
    };

    fetchPrices();
    const interval = setInterval(fetchPrices, 10000);

    return () => clearInterval(interval);
  }, []);

  return { prices, loading };
}
